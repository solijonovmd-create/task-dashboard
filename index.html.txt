<!DOCTYPE html>
<html>
<head>
    <title>Task Dashboard</title>
    <style>
        body { font-family: sans-serif; max-width: 600px; margin: 0 auto; padding: 20px; }
        .task { display: flex; align-items: center; gap: 10px; padding: 8px; border-bottom: 1px solid #eee; }
        .completed { text-decoration: line-through; color: gray; }
        input, button { padding: 8px; }
        .filters { margin: 20px 0; }
    </style>
</head>
<body>
    <h1>Task Dashboard</h1>
    
    <div>
        <input id="taskInput" placeholder="New task..." />
        <button onclick="addTask()">Add</button>
    </div>

    <div class="filters">
        <button onclick="filterAll()">All</button>
        <button onclick="filterActive()">Active</button>
        <button onclick="filterCompleted()">Completed</button>
    </div>

    <div id="taskList"></div>

    <script>
        let tasks = JSON.parse(localStorage.getItem('tasks')) || [];
        let currentFilter = 'all';

        function save() {
            localStorage.setItem('tasks', JSON.stringify(tasks));
            render();
        }

        function render() {
            let filtered = tasks;
            if (currentFilter === 'active') filtered = tasks.filter(t => !t.completed);
            if (currentFilter === 'completed') filtered = tasks.filter(t => t.completed);
            
            const html = filtered.map(task => `
                <div class="task">
                    <input type="checkbox" ${task.completed ? 'checked' : ''} 
                        onclick="toggleTask('${task.id}')">
                    <span class="${task.completed ? 'completed' : ''}">${task.title}</span>
                    <button onclick="deleteTask('${task.id}')">✕</button>
                </div>
            `).join('');
            document.getElementById('taskList').innerHTML = html;
        }

        function addTask() {
            const input = document.getElementById('taskInput');
            if (!input.value.trim()) return alert('Task cannot be blank');
            tasks.push({
                id: Date.now() + '',
                title: input.value,
                completed: false
            });
            input.value = '';
            save();
        }

        function toggleTask(id) {
            const task = tasks.find(t => t.id === id);
            if (task) task.completed = !task.completed;
            save();
        }

        function deleteTask(id) {
            tasks = tasks.filter(t => t.id !== id);
            save();
        }

        function filterAll() { currentFilter = 'all'; render(); }
        function filterActive() { currentFilter = 'active'; render(); }
        function filterCompleted() { currentFilter = 'completed'; render(); }

        render();
    </script>
</body>
</html>
